cmake_minimum_required(VERSION 3.22)
project(adb_pairing_crypto_contract LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT DEFINED BSSL_PREFAB_ROOT)
  message(FATAL_ERROR "Must provide audited BoringSSL prefab root")
endif()
set(BSSL_INCLUDE "${BSSL_PREFAB_ROOT}/prefab/modules/crypto_static/include")
set(BSSL_LIB "${BSSL_PREFAB_ROOT}/prefab/modules/crypto_static/libs/android.${ANDROID_ABI}/libcrypto_static.a")
if(NOT EXISTS "${BSSL_LIB}")
  message(FATAL_ERROR "BoringSSL library missing for ${ANDROID_ABI}")
endif()
add_library(audited_crypto STATIC IMPORTED)
set_target_properties(audited_crypto PROPERTIES IMPORTED_LOCATION "${BSSL_LIB}" INTERFACE_INCLUDE_DIRECTORIES "${BSSL_INCLUDE}")
add_executable(adb_pairing_crypto_contract spake2_crypto_contract.cpp)
target_link_libraries(adb_pairing_crypto_contract PRIVATE audited_crypto log)
target_link_options(adb_pairing_crypto_contract PRIVATE "-Wl,-z,max-page-size=16384")
# Source-only experimental JNI. Not linked into the production APK until audit.
add_library(8bercue_adb_pairing SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../../app/src/release/cpp/on_device_adb_pairing.cpp")
target_link_libraries(8bercue_adb_pairing PRIVATE audited_crypto log)
target_link_options(8bercue_adb_pairing PRIVATE "-Wl,-z,max-page-size=16384")
